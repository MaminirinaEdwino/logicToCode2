mux.HandleFunc("GET /test/{id}/{username}", func(w http.ResponseWriter, r *http.Request) {
	db := config.ConnectDB()
	defer db.Close()

	// Variables d'URL (Path Values)
	id := r.PathValue("id")

	type returnType struct {
		Id int `json:"id"`
		Mark string `json:"mark"`
		Number string `json:"number"`
	}

	var returnValue returnType
	query := "select id, mark, number from voiture where id = $1"
	err := db.QueryRow(query, id).Scan(&returnValue.Id, &returnValue.Mark, &returnValue.Number)
	if err != nil {
		http.Error(w, "model introuvable", http.StatusNotFound)
		return
	}

	renderTemplate(w, ".html", map[string]interface{}{
		"User_table": returnValue,
		"Id": id,
	})
	w.WriteHeader(http.StatusOK)
	renderTemplate(w, ".html", map[string]interface{}{
		"User_table": returnValue,
		"Id": id,
	})
})
