mux.HandleFunc("GET /test/{id}/{username}", func(w http.ResponseWriter, r *http.Request) {
	db := config.ConnectDB()
	defer db.Close()

	// Lecture des paramètres d'URL
	id := r.PathValue("id")

	type returnType struct {
		Id int `json:"id"`
		Username string `json:"username"`
		Password string `json:"password"`
		Email string `json:"email"`
		Role string `json:"role"`
	}

	var returnValue returnType
	query := "select id, username, password, email, role from User_table where id = $1"
	err := db.QueryRow(query, id).Scan(&returnValue.Id, &returnValue.Username, &returnValue.Password, &returnValue.Email, &returnValue.Role)
	if err != nil {
		http.Error(w, "model introuvable", http.StatusNotFound)
		return
	}

	renderTemplate(w, ".html", map[string]interface{}{
		"User_table": returnValue,
		"Id": id,
	})
})
