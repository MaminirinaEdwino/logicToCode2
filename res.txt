mux.HandleFunc("GET /voitures/{id}", func(w http.ResponseWriter, r *http.Request) {
	db := config.ConnectDB()
	defer db.Close()

	// Point d'entrée de la route
	userId := 1
	// Exécution SELECT sur le modèle voiture
	type RowResult struct {
		Mark string `json:"mark"`
		Number string `json:"number"`
	}
	var result RowResult
	query := "SELECT mark, number FROM voiture WHERE id = $1"
	err := db.QueryRow(query, userId).Scan(&result.Mark, &result.Number)
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(result)
})
