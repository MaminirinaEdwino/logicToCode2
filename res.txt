mux.HandleFunc("POST /voitures", func(w http.ResponseWriter, r *http.Request) {
	db := config.ConnectDB()
	defer db.Close()

	// Extraction des paramètres d'URL
	id := r.PathValue("id")

	// --- Modèle cible : voiture ---
	// Action CREATE (INSERT)
	type createInput struct {
		Mark string `json:"mark"`
		Number string `json:"number"`
	}
	var input createInput
	if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
		http.Error(w, "Payload invalide", http.StatusBadRequest)
		return
	}

	// Exécution SQL selon le mode d'opération
	query := "INSERT INTO voiture (mark, number) VALUES ($1, $2) RETURNING id"
	var newID int
	err := db.QueryRow(query, input.Mark, input.Number).Scan(&newID)
	if err != nil {
		http.Error(w, "Erreur lors de l'exécution de la requête", http.StatusInternalServerError)
		return
	}

	w.WriteHeader(http.StatusCreated)
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]any{"id": newID, "message": "Créé avec succès"})
})
