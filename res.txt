mux.HandleFunc("GET /voitures/{id}", func(w http.ResponseWriter, r *http.Request) {
	db := config.ConnectDB()
	defer db.Close()

	// Extraction des paramètres d'URL (Path Values)
	id := r.PathValue("id")

	// --- Modèle cible : voiture ---
	// Sélection configurée sur les champs : [id, mark]
	// Clause d'exclusion / Filtre (WHERE)
	// Structure de retour générée
	type returnType struct {
		Id int `json:"id"`
		Mark string `json:"mark"`
	}

	var returnValue returnType
	query := "SELECT id, mark FROM voiture WHERE id = $1"
	err := db.QueryRow(query, id).Scan(&returnValue.Id, &returnValue.Mark)
	if err != nil {
		http.Error(w, "Ressource introuvable", http.StatusNotFound)
		return
	}

	w.WriteHeader(http.StatusOK)
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(returnValue)
})
