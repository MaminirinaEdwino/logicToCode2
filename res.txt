mux.HandleFunc("GET /voitures/{id}", func(w http.ResponseWriter, r *http.Request) {
	db := config.ConnectDB()
	defer db.Close()

	// Extraction des paramètres d'URL
	id := r.PathValue("id")

	// --- Modèle cible : voiture ---
	// Action SELECT sur : [id, mark]
	// Clause d'exclusion / Filtre dynamique (WHERE)
	// Exécution SQL selon le mode d'opération
	type returnType struct {
		Id int `json:"id"`
		Mark string `json:"mark"`
	}

	var returnValue returnType
	query := "SELECT id, mark FROM voiture WHERE id = $1"
	err := db.QueryRow(query, id).Scan(&returnValue.Id, &returnValue.Mark)
	if err != nil {
		http.Error(w, "Erreur lors de l'exécution de la requête", http.StatusInternalServerError)
		return
	}

	w.WriteHeader(http.StatusOK)
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(returnValue)
})
