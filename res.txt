mux.HandleFunc("POST /test/create", func(w http.ResponseWriter, r *http.Request) {
	db := config.ConnectDB()
	defer db.Close()

	// Reading path parameters if present
	id := r.PathValue("id")

	type inputType struct {
		Id int `json:"id"`
		Username string `json:"username"`
		Password string `json:"password"`
		Email string `json:"email"`
		Role string `json:"role"`
	}

	var inputData inputType
	if err := json.NewDecoder(r.Body).Decode(&inputData); err != nil {
		http.Error(w, "données invalides", http.StatusBadRequest)
		return
	}

	query := "insert into User_table (username, password, email, role) values ($1, $2, $3, $4) returning id"
	var newID int
	err := db.QueryRow(query, inputData.Username, inputData.Password, inputData.Email, inputData.Role).Scan(&newID)
	if err != nil {
		http.Error(w, "erreur lors de la création", http.StatusInternalServerError)
		return
	}

	renderTemplate(w, ".html", map[string]interface{}{
		"Id": newID,
		"Message": "Création réussie",
	})
})
